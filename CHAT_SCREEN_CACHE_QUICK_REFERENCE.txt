===============================================================================
STORAGE KEYS
===============================================================================

Group Chats:
messages_group_{chatId}

Example: messages_group_5

One-to-One Chats:
messages_user_{chatId}_{currentUserId}

Example: messages_user_10_1

===============================================================================
HOW IT WORKS
===============================================================================

FIRST OPEN:
User opens chat → Check cache (empty) → Call API → Save to cache → Display

SUBSEQUENT OPENS:
User opens chat → Load from cache (<100ms) → Display instantly → No API call

QUICK API REFERENCE
===============================================================================

// Check if chat has cached messages
final hasCache = await _syncService.hasCachedMessages(
  chatId, chatType, currentUserId: userId
);

// Load cached messages (instant)
final messages = await _syncService.getCachedMessages(
  chatId, chatType, currentUserId: userId
);

// Load from API (only if no cache)
final result = await _syncService.loadMessagesFromApi(
  chatId: chatId,
  chatType: chatType,
  apiService: apiService,
  currentUserId: userId,
);

// Add message to cache
await _syncService.addMessageToCache(
  chatId, chatType, message, currentUserId: userId
);

// Clear cache for chat
await _syncService.clearCache(
  chatId, chatType, currentUserId: userId
);

// Force reload from API
await _syncService.forceReload(
  chatId, chatType, apiService, currentUserId: userId
);

// Get cache statistics
final stats = await _syncService.getCacheStats(
  chatId, chatType, currentUserId: userId
);
