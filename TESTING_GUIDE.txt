TEST 4: BATCH MESSAGE SEND API
===============================================================================

PURPOSE: Verify batch send works for offline queue

SETUP:
Create test file: test_batch_send.dart

```dart
import 'package:dio/dio.dart';
import 'package:flutter_test/flutter_test.dart';
import '../lib/core/services/api_service_simple.dart';

void main() {
  test('Batch send messages', () async {
    final apiService = ApiService(Dio());
    apiService.setAuthToken('your_test_token');
    
    final messages = [
      {
        'temp_id': 'test_001',
        'chat_id': '1',
        'chat_type': 'group',
        'type': 'text',
        'message': 'Test message 1',
        'firebase_key': 'firebase_test_001',
      },
      {
        'temp_id': 'test_002',
        'chat_id': '1',
        'chat_type': 'group',
        'type': 'text',
        'message': 'Test message 2',
        'firebase_key': 'firebase_test_002',
      },
    ];
    
    final response = await apiService.batchSendMessages(messages);
    
    expect(response['success'], true);
    expect(response['data']['total_sent'], 2);
    expect(response['data']['total_failed'], 0);
    
    print('✅ Batch send test passed');
  });
}
```

RUN:
```bash
flutter test test/test_batch_send.dart
```

EXPECTED DEBUG LOGS:
```

📤 Sample: test_001
📥 Response status: 200
✅ Batch Send Success:
   → Sent: 2 messages
   → Failed: 0 messages

void main() {
  test('Incremental message sync', () async {
    final apiService = ApiService(Dio());
    apiService.setAuthToken('your_test_token');
    
    final lastSync = DateTime.now().subtract(Duration(hours: 1));
    
    final response = await apiService.syncMessages(
      chatId: '1',
      chatType: 'group',
      since: lastSync,
      limit: 100,
    );
    
    expect(response['success'], true);
    expect(response['data']['messages'], isNotNull);
    expect(response['data']['deleted_message_ids'], isNotNull);
    expect(response['data']['has_more'], isNotNull);
    
    print('✅ Incremental sync test passed');
    print('   New messages: ${response['data']['messages'].length}');
    print('   Deleted IDs: ${response['data']['deleted_message_ids'].length}');
  });
}
```
