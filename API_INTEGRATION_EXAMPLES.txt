API 1: BATCH MESSAGE SEND (Offline Queue Sync)
================================================================================

PURPOSE: Send multiple queued messages in one request after coming back online

USAGE EXAMPLE:
--------------
```dart
import 'package:dio/dio.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

// Example: Send 3 queued messages (2 text, 1 image)
final apiService = ApiService(Dio());
apiService.setAuthToken('your_token_here');

final queuedMessages = [
  {
    'temp_id': 'temp_1705315800000',
    'chat_id': '1',
    'chat_type': 'group',
    'type': 'text',
    'message': 'Hello sent offline',
    'firebase_key': 'firebase_key_abc123',
    'client_timestamp': '2024-01-15T10:30:00Z',
  },
  {
    'temp_id': 'temp_1705315900000',
    'chat_id': '2',
    'chat_type': 'user',
    'type': 'text',
    'message': 'Private message',
    'firebase_key': 'firebase_key_def456',
    'reply_to_message_id': 450,  // Reply to previous message
  },
  {
    'temp_id': 'temp_1705316000000',
    'chat_id': '1',
    'chat_type': 'group',
    'type': 'image',
    'message': 'Check this photo',
    'firebase_key': 'firebase_key_ghi789',
    'file_path': 'uploads/images/img_123.jpg',
    'file_name': 'photo.jpg',
    'file_size': 245678,
  },
];

try {
  final response = await apiService.batchSendMessages(queuedMessages);
  
  if (response['success'] == true) {
    final sent = response['data']['sent'] as List;
    final failed = response['data']['failed'] as List;
    
    print('✅ Sent: ${sent.length} messages');
    print('❌ Failed: ${failed.length} messages');
    
    // Update local database with server IDs
    for (final msg in sent) {
      final tempId = msg['temp_id'];
      final serverId = msg['id'];
      // Update Hive: Replace temp_id with server id
      await updateMessageId(tempId, serverId);
    }
  }
} catch (e) {
  print('Error: $e');
}
```

API 2: INCREMENTAL MESSAGE SYNC
================================================================================

PURPOSE: Fetch only new/updated/deleted messages since last sync (saves bandwidth)

USAGE EXAMPLE:
--------------
```dart
// Get last sync time from Hive
final lastSync = await hiveBox.get('last_sync_group_1') ?? 
                 DateTime.now().subtract(Duration(days: 7));

try {
  final response = await apiService.syncMessages(
    chatId: '1',
    chatType: 'group',
    since: lastSync,
    limit: 100,
  );
  
  if (response['success'] == true) {
    final data = response['data'];
    final newMessages = data['messages'] as List;
    final deletedIds = data['deleted_message_ids'] as List;
    final updatedMessages = data['updated_messages'] as List;
    
    print('📥 New: ${newMessages.length}');
    print('🗑️ Deleted: ${deletedIds.length}');
    print('✏️ Updated: ${updatedMessages.length}');
    
    // Update local Hive database
    for (final msg in newMessages) {
      await saveMessageToHive(Message.fromJson(msg));
    }
    
    for (final msgId in deletedIds) {
      await deleteMessageFromHive(msgId);
    }
    
    for (final msg in updatedMessages) {
      await updateMessageInHive(Message.fromJson(msg));
    }
    
    // Save sync timestamp
    await hiveBox.put('last_sync_group_1', data['synced_at']);
  }
} catch (e) {
  print('Error: $e');
}
```

API 3: INCREMENTAL CHAT LIST SYNC
================================================================================

PURPOSE: Sync only changed chats (new/updated/deleted) instead of full list

USAGE EXAMPLE:
--------------
```dart
// Get last chat list sync time
final lastChatSync = await hiveBox.get('last_chat_sync') ?? 
                     DateTime.now().subtract(Duration(days: 1));

try {
  final response = await apiService.syncChats(
    since: lastChatSync,
    limit: 50,
  );
  
  if (response['success'] == true) {
    final data = response['data'];
    final newChats = data['new_chats'] as List;
    final updatedChats = data['updated_chats'] as List;
    final deletedIds = data['deleted_chat_ids'] as List;
    
    print('🆕 New chats: ${newChats.length}');
    print('🔄 Updated chats: ${updatedChats.length}');
    print('🗑️ Deleted chats: ${deletedIds.length}');
    
    // Update Hive
    for (final chat in newChats) {
      await saveChatToHive(ChatHiveModel.fromApi(chat));
    }
    
    for (final chat in updatedChats) {
      await updateChatInHive(ChatHiveModel.fromApi(chat));
    }
    
    for (final chatId in deletedIds) {
      await deleteChatFromHive(chatId);
    }
    
    // Save sync time
    await hiveBox.put('last_chat_sync', data['synced_at']);
  }
} catch (e) {
  print('Error: $e');
}
```

API 5: QUEUE STATUS CHECK
================================================================================

PURPOSE: Check how many messages are pending/failed in offline queue

USAGE EXAMPLE:
--------------
```dart
try {
  final response = await apiService.getQueueStatus();
  
  if (response['success'] == true) {
    final pending = response['data']['pending_messages'];
    final uploads = response['data']['pending_uploads'];
    final failed = response['data']['failed_messages'];
    final isFull = response['data']['is_queue_full'];
    final maxSize = response['data']['max_queue_size'];
    
    print('📊 Queue Status:');
    print('   Pending: $pending');
    print('   Uploads: $uploads');
    print('   Failed: $failed');
    print('   Capacity: $pending/$maxSize ${isFull ? "(FULL)" : ""}');
    
    // Show warning if queue is full
    if (isFull) {
      showWarning('Message queue full. Please connect to internet.');
    }
  }
} catch (e) {
  print('Error: $e');
}
```

EXPECTED RESPONSE:
```json
{
  "success": true,
  "data": {
    "pending_messages": 12,
    "pending_uploads": 3,
    "failed_messages": 0,
    "oldest_pending_timestamp": "2024-01-15T10:30:00+00:00",
    "queue_size_mb": 2.5,
    "is_queue_full": false,
```

API 6: APP VERSION CHECK
================================================================================

PURPOSE: Check for app updates (force/optional) without authentication

USAGE EXAMPLE:
--------------
```dart
import 'package:package_info_plus/package_info_plus.dart';

// Get current app version
final packageInfo = await PackageInfo.fromPlatform();
final currentVersion = packageInfo.version; // e.g., "1.0.23"

try {
  // No auth token required for this API
  final response = await apiService.checkAppVersion(
    currentVersion: currentVersion,
  );
  
  if (response['success'] == true) {
    final data = response['data'];
    final updateAvailable = data['update_available'];
    
    if (updateAvailable) {
      final updateType = data['update_type']; // "force" or "optional"
      final latestVersion = data['latest_version'];
      final releaseNotes = data['release_notes'];
      final downloadUrl = data['download_url'];
      
      if (updateType == 'force') {
        // Show blocking dialog
        showForceUpdateDialog(
          message: data['force_update_message'],
          downloadUrl: downloadUrl,
        );
      } else {
        // Show optional dialog
        showOptionalUpdateDialog(
          currentVersion: currentVersion,
          latestVersion: latestVersion,
          releaseNotes: releaseNotes,
          downloadUrl: downloadUrl,
        );
      }
    } else {
      print('✅ App is up to date');
    }
  }
} catch (e) {
  print('Error: $e');
}
```
